-- 1. Create applications table for multi-stage onboarding workflow
create table if not exists public.applications (
  id uuid default uuid_generate_v4() primary key,
  profile_id uuid references public.profiles(id) on delete cascade unique,
  first_name text not null,
  last_name text not null,
  email text not null unique,
  phone_number text not null,
  country text not null,
  track text check (track in ('Product design', 'Frontend developer', 'backend developer', 'product manager', 'Business analyst', 'project coordinator', 'project manager')),
  cohort_id uuid references public.cohorts(id) on delete set null,
  cohort_name text, -- e.g. "Cohort 4 - Match, 2027"
  experience_level text check (experience_level in ('Beginner', 'Some Experience', 'Experienced')),
  portfolio_link text,
  linkedin_link text,
  resume_url text,
  why_join text,
  payment_method text not null default 'Paystack',
  payment_status text not null default 'Pending' check (payment_status in ('Pending', 'Successful', 'Failed')),
  reference_code text unique, -- e.g. #PNA-2026-08341
  application_status text not null default 'Draft' check (application_status in ('Draft', 'Submitted', 'Under Review', 'Accepted', 'Rejected')),
  current_stage integer not null default 1, -- 1: Personal, 2: Program, 3: Experience, 4: Review & Payment
  created_at timestamp with time zone default timezone('utc'::text, now()) not null,
  updated_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 5. Create a function to automatically sync new sign-ups from auth.users to public.profiles
create or replace function public.handle_new_user()
returns trigger
language plpgsql
security definer set search_path = ''
as $$
begin
  insert into public.profiles (id, first_name, last_name, email, role)
  values (
    new.id,
    new.raw_user_meta_data ->> 'first_name',
    new.raw_user_meta_data ->> 'last_name',
    new.email,
    coalesce(new.raw_user_meta_data ->> 'role', 'Applicant')
  );
  return new;
end;
$$;

-- 6. Create a trigger that fires every time a new user registers in Supabase Auth
drop trigger if exists on_auth_user_created on auth.users;
create trigger on_auth_user_created
  after insert on auth.users
  for each row execute procedure public.handle_new_user();
